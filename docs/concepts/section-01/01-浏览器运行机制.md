### 1.
```js

console.log('4');
```

- 输出顺序为： 1 - 4 - 3 -2

- 输出顺序为 3 - 1 - 4 - 2;

### 3.

### 4.
- await 后面跟任何非 Promise 值，引擎都会自动包装成 Promise.resolve(值)。 
```js
async function foo() {
  console.log('1');
  await 123;
  console.log('2');
}

foo();
console.log('3');
```

- 输出顺序为： 1 - 3 - 2

### 5.
- queueMicrotask 和 Promise.then 走的是同一个微任务队列，按注册顺序排列。
- queueMicrotask 是显式入队微任务的标准 API，比 Promise.resolve().then(fn) 更语义化，且不需要创建多余的 Promise 对象。

```js
console.log('1');

queueMicrotask(() => console.log('2'));

Promise.resolve().then(() => console.log('3'));

queueMicrotask(() => console.log('4'));

console.log('5');
```
- 输出顺序为： 1 - 5 - 2 - 3 - 4

### 6.
- 如果在微任务里不断注册新的微任务，队列永远清不空，渲染和宏任务永远等不到。

```js
// 危险示例：会卡死页面
let i = 0;
function loop() {
  console.log(i++);
  Promise.resolve().then(loop); // 每次都注册新微任务
}
Promise.resolve().then(loop);
// 浏览器会一直输出数字，永远不渲染，永远不响应输入
```
- 如果在 Promise.then 链里做推理循环且不 yield，就会出现这种情况 —— 推理一直在跑，UI 完全卡死

### 7.
- setTimeout(fn, 0) 不是真的 0ms，浏览器有 最小 4ms 延迟（当嵌套层级 ≥ 5 时）
- 即使计时到了，也必须等当前同步代码 + 所有微任务执行完才会执行
- 未激活的标签页中，setTimeout 会被节流到至少 1000ms

```js
console.log('1');

setTimeout(() => {
  console.log('2');
  setTimeout(() => console.log('3'), 0);
}, 0);

setTimeout(() => console.log('4'), 0);

console.log('5');
```

- 输出顺序为： 1 - 5 - 2 - 4 - 3

    - 只在需要渲染时执行（通常每帧一次，约 16.6ms）
    - 标签页不可见时，rAF 会暂停
    - 适合做动画，不适合做精确计时

- 输出顺序为： 1 - 6 - 5 - 3 - 4 - 2  ✅

## 宏任务、微任务分类

微任务： .then() 、 await之后所有的代码会被包装成微任务、 queueMicrotask
宏任务： 事件回调（如 onload() 、 addEventListener等）、 所有定时器相关的操作、 file.text() / file.arrayBuffer() （返回pPomise）、 response.body.getReader().read() 【(Streams API 返回 Promise)】、  response.blob() / response.json()

* Q1: 为什么同步执行多个axios请求或者调用接口的请求时，接口返回的信息不是按顺序执行的？按上面所说，接口的响应通过.then获取属于微任务，微任务应该按顺序进入微任务队列，但为什么返回的接口不是按调用的顺序返回的呢？

* A: 请求本身不分宏微,都是同步执行： 核心原因是：.then() 回调不是在调用 .then() 时入队的，而是在 Promise 被 resolve 的那一刻才入队。 谁先 resolve，谁的 .then() 先进队列，和发起请求的顺序无关

## Promise 相关巩固

### 执行顺序： 
```js
console.log('1');

const p = new Promise((resolve, reject) => {
  console.log('2');  // 同步执行！
  resolve('3');      // 异步触发 .then
  console.log('4');  // 同步执行！
});

console.log('5');
```
- 执行顺序为 1 - 2 - 4 - 5 - 3
- 说明： new Promise(executor)中， executor 立刻同步执行，不是微任务。 .then() / .catch() 里的代码才是微任务。

### .then()的返回规则

```js
Promise.resolve(1)
  .then(v => v * 2)           // return 2 → fulfilled(2)
  .then(v => {
    return Promise.resolve(v * 3);  // return Promise(6) → 采用它
  })
  .then(v => {
    throw new Error('boom');   // throw → rejected
  })
  .then(
    v => console.log('不会执行'),
    e => console.log('捕获:', e.message)  // 第二个参数也能捕获
  );
// 输出：捕获: boom
```

### .catch()的错误传递机制
```js
Promise.resolve()
  .then(() => {
    throw new Error('第一步出错');
  })
  .then(() => console.log('不会执行'))   // 跳过
  .then(() => console.log('也不会执行')) // 跳过
  .catch(err => {
    console.log('捕获到:', err.message); // 在这里被捕获
    return '恢复正常';                    // catch 也返回新 Promise
  })
  .then(v => console.log('继续执行:', v)); // 输出：继续执行: 恢复正常
```

### Promise的静态方法
- Promise.all(): 按顺序的结果数组, 任一失败 → 立即 reject

```js
const [user, orders, products] = await Promise.all([
  fetchUser(1),
  fetchOrders(1),
  fetchProducts(),
]);
```
- 特性： 
    - 三个请求并发，总耗时 ≈ 最慢的那个
    - 所有 Promise 同时开始
    - 结果数组顺序和输入顺序一致，和完成顺序无关



- Promise.race(): 返回最先成功或失败的值（通常用于超时控制）
```js
// 给请求加超时：3秒没返回就超时
function withTimeout(promise, ms) {
  return Promise.race([
    promise,
    new Promise((_, reject) =>
      setTimeout(() => reject(new Error('超时')), ms)
    ),
  ]);
}

try {
  const data = await withTimeout(fetch('/api/slow'), 3000);
} catch (err) {
  console.log(err.message); // '超时'
}
```

- Promise.allSettled（全部完成并返回，不论成功失败）
```js
const results = await Promise.allSettled([
  Promise.resolve('成功A'),
  Promise.reject('失败B'),
  Promise.resolve('成功C'),
]);

results.forEach(r => {
  if (r.status === 'fulfilled') {
    console.log('成功:', r.value);
  } else {
    console.log('失败:', r.reason);
  }
});
// 成功: 成功A
// 失败: 失败B
// 成功: 成功C
```

- Promise.any (返回第一个成功的值， 全部失败才 reject（AggregateError）)
```js
try {
  const result = await Promise.any([
    Promise.reject('失败1'),
    Promise.resolve('成功2'),
    Promise.reject('失败3'),
  ]);
  console.log(result); // 成功2
} catch (err) {
  console.log(err.errors); // 全部失败时是 AggregateError
}
```
- Promise.resolve()
- Promise.reject()

## async / await 相关巩固
- 本质还是Promise的语法糖

```js
// Promise 写法
fetchUser(1)
  .then(user => fetchOrders(user.id))
  .then(orders => console.log(orders))
  .catch(err => console.error(err));

// async/await 写法（等价）
async function loadData() {
  try {
    const user = await fetchUser(1);
    const orders = await fetchOrders(user.id);
    console.log(orders);
  } catch (err) {
    console.error(err);
  }
}
```

### async 函数的本质
- 永远返回一个 Promise
```js
async function foo () {
    return 42;
}

const result = foo();
console.log(result instanceof Promise); // true
result.then(v => console.log(v));  // 42
```

### await 的本质
- await 会把后面的代码变成微任务
```js
async function loadData() {
  try {
    const user = await fetchUser(1);
    const orders = await fetchOrders(user.id);
    return orders;
  } catch (err) {
    // 任何一个 await 失败都会跳到这里
    console.error('加载失败:', err);
    return [];  // catch 里 return 会变成 Promise.resolve([])
  } finally {
    setLoading(false); // 无论成败都执行
  }
}

// 如果不写 try/catch，async 函数内的 await 失败会导致整个 async 函数返回的 Promise 被 reject，需要在调用处 .catch()：
async function risky() {
  await Promise.reject(new Error('输出报错'));
}

risky().catch(err => console.log(err.message)); // 输出报错
// 不写 .catch() 会报 Uncaught (in promise)
```

### *并行 / 串行 （性能关键）
```js
// 串行： 顺序执行， 慢
async function serial() {
    const a = await fetchA(); // 100 ms
    const b = await fetchB(); // 200 ms(等A完成开始)
    const c = await fetchC(); // 150 ms（等B完成开始）
}
// 总耗时 = 100 + 200 + 150 = 450 ms

// 并行： 并发执行
async function parallel() {
    const [a, b, c] = await Promise.all([
        fetchA(),
        fetchB(),
        fetchC()
    ])
}
// 总耗时 ≈ 200 ms
```

#### 半并行模式
```js
async function semiParallel() {
    const user = await fetchUser(1);
    const [order, profile] = await Promise.all([
        fetchOrders(user.id),    // 依赖 user
        fetchProfile(user.id),   // 依赖 user，但两者互相独立
    ])
}
```
